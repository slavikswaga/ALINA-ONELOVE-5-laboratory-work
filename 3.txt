#include <string>
#include <iostream>
#include <iomanip>
#include <stdint.h>
#include <bitset>


static std::string bin(uint16_t x) {
	uint16_t i = ~((uint16_t)(-1) >> 1), p = 0;
	std::string r = "";
	while (i != 0) {
		if (x & i) r += '1';
		else     r += '0';
		i = i >> 1;
		p++;
		if (p % 4 == 0) r += " ";
	}
	return r;
}

void ShowTitle() {
	std::cout << "\t| dec   | oct  | hex| bin\n";
}
void ShowValue(std::string beg, uint16_t x) { //**
	std::cout << beg << "\t|"
		<< std::dec << std::setw(7) << std::setfill(' ') << std::right << x << "|"
		<< std::oct << std::setw(6) << std::setfill('0') << std::internal << x << "|"
		<< std::hex << std::setw(4) << std::setfill('0') << std::internal << x << "| "
		<< bin(x) << std::endl;
}

int main() {
	setlocale(LC_ALL, "RU");
	int16_t x;
	std::cout << "Введите число: ";
	std::cin >> x;

	uint16_t IsNegat = (x >> 15) & 1;
	IsNegat = (IsNegat << 15);

	uint16_t absx = (x < 0) ? -x : x;

	uint16_t PKx = IsNegat | absx;
	uint16_t OBRKx, DKx;
	if (x < 0) {
		OBRKx = ~PKx | 0b1000000000000000;
		DKx = OBRKx + 1;
	}
	else {
		OBRKx = PKx; //если число положительное то ОК и ДК совпадают в ПК
		DKx = PKx;
	}

	ShowTitle();
	ShowValue("ПК - ", PKx);
	ShowValue("ОК - ", OBRKx);
	ShowValue("ДК - ", DKx);
	system("pause");
	return 0;
}
